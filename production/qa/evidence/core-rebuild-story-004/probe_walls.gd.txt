extends SceneTree
# Wall at x=0 facing +X (one-sided), y in [0,4], z in [-5,5]. Sphere R=1.
var st: PhysicsDirectSpaceState3D
var q := PhysicsShapeQueryParameters3D.new()
var cr := PhysicsCastMotionResult3D.new()
var rr := PhysicsGetRestInfoResult3D.new()
var sph := SphereShape3D.new()

func _cast(from: Vector3, to: Vector3) -> String:
	q.transform = Transform3D(Basis.IDENTITY, from)
	q.motion = to - from
	var ok := st.cast_motion_into(q, cr)
	q.motion = Vector3.ZERO
	var s := "cast %s->%s ret=%s safe=%.6f unsafe=%.6f" % [from, to, ok, cr.safe_fraction, cr.unsafe_fraction]
	q.transform = Transform3D(Basis.IDENTITY, to)
	if st.get_rest_info_into(q, rr):
		s += " rest@to p=%s n=%s shape=%d" % [rr.point, rr.normal, rr.collider_shape]
	else:
		s += " rest@to none"
	return s

func _initialize() -> void:
	var space := PhysicsServer3D.space_create()
	st = PhysicsServer3D.space_get_direct_state(space)
	var shape := PhysicsServer3D.concave_polygon_shape_create()
	var a := Vector3(0, 0, -5); var b := Vector3(0, 4, -5); var c := Vector3(0, 0, 5); var d := Vector3(0, 4, 5)
	PhysicsServer3D.shape_set_data(shape, {"faces": PackedVector3Array([a, c, b, c, d, b]), "backface_collision": false})
	var body := PhysicsServer3D.body_create()
	PhysicsServer3D.body_set_mode(body, PhysicsServer3D.BODY_MODE_STATIC)
	PhysicsServer3D.body_set_collision_layer(body, 2)
	PhysicsServer3D.body_set_collision_mask(body, 0)
	PhysicsServer3D.body_set_space(body, space)
	PhysicsServer3D.body_add_shape(body, shape)
	sph.radius = 1.0
	q.shape = sph
	q.collision_mask = 2
	print("margin=", q.margin)
	print(_cast(Vector3(3, 2, 0), Vector3(1.0, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(0.999, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(0.99, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(0.9, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(1.001, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(3, 2, 0)))
	print(_cast(Vector3(3, 2, 0), Vector3(-3, 2, 0)))
	print(_cast(Vector3(-3, 2, 0), Vector3(3, 2, 0)))
	print(_cast(Vector3(3, 2, -2), Vector3(1.0, 2, 2)))
	sph.radius = 0.5
	print("r=0.5 ", _cast(Vector3(3, 2, 0), Vector3(0.9, 2, 0)))
	print("r=0.5 ", _cast(Vector3(3, 2, 0), Vector3(0.5, 2, 0)))
	PhysicsServer3D.free_rid(body)
	PhysicsServer3D.free_rid(shape)
	PhysicsServer3D.free_rid(space)
	quit()
